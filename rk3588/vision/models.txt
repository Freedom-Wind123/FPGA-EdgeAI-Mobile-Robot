# 多模型并行配置：每行一个模型
# name route model classes core every conf parent
#   name    : 模型名，日志与去重用
#   route   : yolov8（80 类主模型）或 split（双输出社区模型）
#   model   : rknn 路径
#   classes : 该模型的类别表，行数必须等于模型输出类别数
#   core    : NPU 核 0/1/2
#   every   : 每 N 帧投递一次，base 必须为 1，其余模型降频省算力
#   conf    : 该模型自己的置信度阈值
#   parent  : 父类名（可含空格，位于行尾；- 表示无父类）
#             parent 前可加 drop：同框 IoU≥0.6 时丢弃父类粗框，只留 split 细类（状态类模型用）
#             require=cls1,cls2 可选：该模型的框必须与这些类别同框才保留（如车位需有自行车）
# 注意：base 必须排在第一条，保证 COCO 类别仍占用统一表 0~79 的下标
base yolov8 models/yolov8n_int8.rknn classes.txt 0 1 0.18 -
community split models/community_non_community.rknn classes_community_people.txt 0 3 0.30 person
fire yolov8 models/fire.rknn classes_fire.txt 1 3 0.30 -
garbage split models/garbage_can_yolo11n.rknn classes_garbage.txt 2 3 0.30 drop trash can
traffic split models/traffic_light_yolo11n.rknn classes_traffic_light.txt 1 5 0.30 drop traffic light
parking split models/bike_parking_yolo11n.rknn classes_bike_parking.txt 2 5 0.30 - require=bicycle,motorcycle
